2) Complete Data Schema Spec
2.1 Canonicalization Rules (Non-negotiable)
Canonical JSON for hashing/signing must follow:

UTF-8 encoding

Stable key order: lexicographic by key at every object level

No floating timestamps: timestamps must be ISO-8601 Zulu with ms (e.g., 2026-01-22T12:34:56.789Z)

No NaN/Infinity values

Explicit null: missing optional fields omitted, not null (except where required)

Arrays preserve order (do not sort unless schema says sorted)

Whitespace: no pretty formatting (minified)

IDs must be strings; never auto-coerce numbers

Hashes are lowercase hex; prefix sha256: etc.

Signature block is computed after event_hash is computed (signature signs the canonical event without integrity.signature field, then signature is inserted)

Hashing

event_hash = sha256(canonical_json_without_signature)

signature = ed25519_sign(event_hash) (or sign canonical bytes)

3) Security Appendix
3.1 Trust boundaries
SDK ↔ Enforcer: authenticated + rate-limited

Enforcer ↔ Evidence: internal mTLS

Evidence ↔ Vault/Store: mTLS + least-privileged creds

Protected Targets verify STIX tokens independently

3.2 Key management
Signing keys (Evidence + Proof Tokens)
Algorithm: Ed25519 (preferred) or ES256 for JWT ecosystems

Storage: KMS/HSM-backed

Key hierarchy options:

Global signing key (MVP)

Per-tenant signing keys (recommended for enterprise)

Key rotation
Maintain key ring: kid identifies active key

Rotation cadence: 90 days (configurable)

Evidence includes integrity.key_id

Token header includes kid

Verification supports old keys until retention requires

3.3 mTLS and workload identity
Internal service-to-service: mTLS with SPIFFE/SPIRE or cloud-native identity

SDK auth: OIDC client credentials or workload identity

Principle of least privilege for vault writer

3.4 Token formats
JWT for broad compatibility OR PASETO for stronger defaults

TTL: 30–60 seconds

Claims must include: actor, action, resource summary, policy hash, trace_id

Tokens must be auditable: every ALLOW token minted corresponds to evidence event

3.5 Incident response requirements
Key compromise

revoke key, rotate, publish incident evidence event

auditors can see impacted period by key_id

STIX outage

fail-closed for high-risk actions

emit operational evidence for outage windows

Vault write failure

degrade to queued durable pipeline

ALLOW cannot be returned unless evidence is durably accepted (queue ack)

3.6 Data protection
Default: do not store raw sensitive payloads

Encrypt at rest (hot store + vault)

Field-level controls:

args_summary must never contain PII/PHI/PCI

classification only

export function loadPolicies(dir: string): PolicyArtifact[] {
  const files = fs.readdirSync(dir).filter(f => f.endsWith(".yaml") || f.endsWith(".yml"));
  const policies: PolicyArtifact[] = [];
  for (const f of files) {
    const p = path.join(dir, f);
    const doc = yaml.load(fs.readFileSync(p, "utf8")) as any;
    policies.push(doc as PolicyArtifact);
  }
  return policies;
}

export function policyHash(policy: PolicyArtifact): string {
  // Stable hash: sha256 over JSON string of policy (keys as inserted by YAML parser; acceptable for skeleton)
  const bytes = Buffer.from(JSON.stringify(policy), "utf8");
  return `sha256:${crypto.createHash("sha256").update(bytes).digest("hex")}`;
}
services/policy/src/evaluator.ts
import { AuthorizeRequest, PolicyArtifact, PolicyDecision } from "./types.js";
import { policyHash } from "./store.js";

function matchScope(req: AuthorizeRequest, p: PolicyArtifact): boolean {
  return req.actor.type === p.scope.actor_type &&
         req.action === p.scope.action &&
         req.resource.type === p.scope.resource_type;
}

function matchConditions(req: AuthorizeRequest, p: PolicyArtifact): boolean {
  const c = p.conditions || {};
  if (c.env && req.context.env !== c.env) return false;

  if (c.require_signal) {
    for (const [k, v] of Object.entries(c.require_signal)) {
      const actual = (req.context.signals as any)?.[k];
      if (actual !== v) return false;
    }
  }

  if (c.require_resource_attr) {
    for (const [k, v] of Object.entries(c.require_resource_attr)) {
      const actual = (req.resource.attributes as any)?.[k];
      if (actual !== v) return false;
    }
  }
  return true;
}

export function evaluate(req: AuthorizeRequest, policies: PolicyArtifact[]): PolicyDecision {
  // Simple: first matching policy wins (extend with priority later)
  for (const p of policies) {
    if (!matchScope(req, p)) continue;
    if (!matchConditions(req, p)) continue;

    return {
      decision: p.decision,
      reason_code: p.reason_code,
      policy: { id: p.policy_id, version: p.version, hash: policyHash(p) },
      obligations: p.obligations || [],
      approval: p.decision === "QUARANTINE"
        ? { required: true, approver_role: p.approval?.approver_role }
        : { required: false },
      controls: p.controls || [],
      regulations: p.regulations || []
    };
  }

  // Default: deny unknown unless explicitly allowed
  return {
    decision: "DENY",
    reason_code: "NO_MATCHING_POLICY_DEFAULT_DENY",
    policy: { id: "DEFAULT", version: "0", hash: "sha256:" + "0".repeat(64) },
    obligations: [],
    approval: { required: false },
    controls: [],
    regulations: []
  };
}

import nacl from "tweetnacl";

function hexToUint8(hex: string): Uint8Array {
  const clean = hex.trim().toLowerCase();
  const bytes = new Uint8Array(clean.length / 2);
  for (let i = 0; i < bytes.length; i++) bytes[i] = parseInt(clean.slice(i * 2, i * 2 + 2), 16);
  return bytes;
}

export function makeSigner(seedHex: string) {
  const seed = hexToUint8(seedHex);
  if (seed.length !== 32) throw new Error("Seed must be 32 bytes hex");
  const kp = nacl.sign.keyPair.fromSeed(seed);

  return {
    signUtf8: (msg: string) => {
      const sig = nacl.sign.detached(Buffer.from(msg, "utf8"), kp.secretKey);
      return Buffer.from(sig).toString("base64");
    },
    publicKeyBase64: Buffer.from(kp.publicKey).toString("base64")
  };
}

app.post("/evidence", (req, res) => {
  const incoming = req.body as any;

  const event_id = incoming.event_id || `evt_${uuidv4()}`;
  const timestamp = incoming.timestamp || new Date().toISOString();

  const prev_hash = loadPrevHash(cfg.hotDir);

  // Build event without signature first
  const event = {
    ...incoming,
    schema_version: incoming.schema_version || "1.0",
    event_id,
    timestamp,
    integrity: {
      event_hash: "sha256:" + "0".repeat(64), // placeholder
      prev_hash,
      signature: "", // placeholder
      key_id: cfg.kid
    }

  // Canonicalize without signature for hashing/signing

    ...event,
    integrity: { ...event.integrity, signature: "" }

  const event_hash = sha256Prefixed(canonicalNoSig);

  // Sign the hash string (simple); production can sign canonical bytes instead
  const signature = signer.signUtf8(event_hash);

  const finalEvent = {
    ...event,
    integrity: {
      ...event.integrity,
      event_hash,
      signature
    }
  };

  const canonicalFinal = canonicalize(finalEvent);

  // Store hot + vault
  appendJsonl(cfg.hotDir, "evidence.jsonl", finalEvent);
  writeVaultObject(cfg.vaultDir, event_id, canonicalFinal);
  savePrevHash(cfg.hotDir, event_hash);

  res.json({ ok: true, event_id, event_hash, prev_hash, key_id: cfg.kid });
});

export class StixGuardrails {
  constructor(
    private stix: StixClient,
    private base: {
      actor: AuthorizeRequest["actor"];
      context: AuthorizeRequest["context"];
    }
  ) {}

  wrapTool<TArgs, TResult>(
    meta: ToolMeta,
    handler: (args: TArgs, proofToken?: string) => Promise<TResult>
  ): (args: TArgs) => Promise<GuardedResult<TResult>> {
    return async (args: TArgs) => {
      const signals = meta.classify ? meta.classify(args) : {};
      const resourceAttrs = meta.deriveResourceAttributes ? meta.deriveResourceAttributes(args) : {};

      const req: AuthorizeRequest = {
        actor: this.base.actor,
        action: meta.action,
        resource: {
          type: meta.resourceType,
          classification: signals?.piiDetected ? ["PII"] : [],
          attributes: resourceAttrs
        },
        context: {
          ...this.base.context,
          signals
        },
        tool_call: {
          tool: meta.name,
          risk_tier: meta.riskTier,
          args_summary: meta.summarizeArgs ? meta.summarizeArgs(args) : undefined
        }
      };

      const decision = await this.stix.authorize(req);

      if (decision.decision === "DENY") return { kind: "DENIED", decision };
      if (decision.decision === "QUARANTINE") return { kind: "QUARANTINED", decision };

      // ALLOW
      const { args: newArgs } = applyObligations(args, decision.obligations || []);
      const result = await handler(newArgs as any, decision.proof_token);
      return { kind: "OK", value: result, trace_id: decision.trace_id };
    };
  }
}

function base64urlToBuf(b64url: string): Buffer {
  const b64 = b64url.replace(/-/g, "+").replace(/_/g, "/") + "===".slice((b64url.length + 3) % 4);
  return Buffer.from(b64, "base64");
}

export function makeStixVerifier(opts: { publicKeyBase64: string; expectedIssuer?: string }) {
  const pub = Buffer.from(opts.publicKeyBase64, "base64");

  return function stixVerify(req: any, res: any, next: any) {
    const auth = req.headers["authorization"] || "";
    const token = typeof auth === "string" && auth.startsWith("Bearer ") ? auth.slice(7) : "";

    if (!token) return res.status(401).json({ error: "missing_stix_token" });

    const parts = token.split(".");
    if (parts.length !== 3) return res.status(401).json({ error: "invalid_token_format" });

    const signingInput = `${parts[0]}.${parts[1]}`;
    const sig = base64urlToBuf(parts[2]);

    const ok = nacl.sign.detached.verify(Buffer.from(signingInput, "utf8"), sig, pub);
    if (!ok) return res.status(401).json({ error: "invalid_signature" });

    const payload = JSON.parse(base64urlToBuf(parts[1]).toString("utf8"));
    const now = Math.floor(Date.now() / 1000);
    if (payload.exp && now > payload.exp) return res.status(401).json({ error: "token_expired" });
    if (opts.expectedIssuer && payload.iss !== opts.expectedIssuer) return res.status(401).json({ error: "wrong_issuer" });

    // attach claims
    req.stix = payload;
    next();
  };
}

import { AuthorizeRequest, PolicyArtifact, PolicyDecision, DecisionKind } from "./types.js";
import { policyHash } from "./store.js";

function matchScope(req: AuthorizeRequest, p: PolicyArtifact): boolean {
  return req.actor.type === p.scope.actor_type &&
         req.action === p.scope.action &&
         req.resource.type === p.scope.resource_type;
}

function matchConditions(req: AuthorizeRequest, p: PolicyArtifact): boolean {
  const c = p.conditions || {};
  if (c.env && req.context.env !== c.env) return false;

  if (c.require_signal) {
    for (const [k, v] of Object.entries(c.require_signal)) {
      const actual = (req.context.signals as any)?.[k];
      if (actual !== v) return false;
    }
  }

  if (c.require_resource_attr) {
    for (const [k, v] of Object.entries(c.require_resource_attr)) {
      const actual = (req.resource.attributes as any)?.[k];
      if (actual !== v) return false;
    }
  }
  return true;
}

function decisionRank(d: DecisionKind): number {
  // deny-wins, then quarantine, then allow
  if (d === "DENY") return 0;
  if (d === "QUARANTINE") return 1;
  return 2; // ALLOW
}

export function evaluate(req: AuthorizeRequest, policies: PolicyArtifact[]): PolicyDecision {
  const matches = policies
    .filter(p => matchScope(req, p) && matchConditions(req, p))
    .map(p => ({
      p,
      priority: typeof p.priority === "number" ? p.priority : 100
    }));

  if (matches.length === 0) {
    return {
      decision: "DENY",
      reason_code: "NO_MATCHING_POLICY_DEFAULT_DENY",
      policy: { id: "DEFAULT", version: "0", hash: "sha256:" + "0".repeat(64) },
      obligations: [],
      approval: { required: false },
      controls: [],
      regulations: []
    };
  }

  // Deterministic sort:
  // 1) priority asc (lower number wins)
  // 2) decision rank (DENY < QUARANTINE < ALLOW)
  // 3) policy_id lexicographic (tie-breaker)
  matches.sort((a, b) => {
    if (a.priority !== b.priority) return a.priority - b.priority;
    const dr = decisionRank(a.p.decision) - decisionRank(b.p.decision);
    if (dr !== 0) return dr;
    return a.p.policy_id.localeCompare(b.p.policy_id);
  });

  const winner = matches[0].p;
  return {
    decision: winner.decision,
    reason_code: winner.reason_code,
    policy: { id: winner.policy_id, version: winner.version, hash: policyHash(winner) },
    obligations: winner.obligations || [],
    approval: winner.decision === "QUARANTINE"
      ? { required: true, approver_role: winner.approval?.approver_role }
      : { required: false },
    controls: winner.controls || [],
    regulations: winner.regulations || []
  };
}

import { StixClient } from "./stixClient.js";
import { AuthorizeRequest, GuardedResult, ToolMeta } from "./types.js";
import { applyObligations } from "./obligations.js";

export class StixGuardrails {
  constructor(
    private stix: StixClient,
    private base: {
      actor: AuthorizeRequest["actor"];
      context: AuthorizeRequest["context"];
    }
  ) {}

  wrapTool<TArgs, TResult>(
    meta: ToolMeta,
    handler: (args: TArgs, proofToken?: string) => Promise<TResult>
  ): (args: TArgs) => Promise<GuardedResult<TResult>> {
    return async (args: TArgs) => {
      const signals = meta.classify ? meta.classify(args) : {};
      const resourceAttrs = meta.deriveResourceAttributes ? meta.deriveResourceAttributes(args) : {};

      const req: AuthorizeRequest = {
        actor: this.base.actor,
        action: meta.action,
        resource: {
          type: meta.resourceType,
          classification: signals?.piiDetected ? ["PII"] : [],
          attributes: resourceAttrs
        },
        context: { ...this.base.context, signals },
        tool_call: {
          tool: meta.name,
          risk_tier: meta.riskTier,
          args_summary: meta.summarizeArgs ? meta.summarizeArgs(args) : undefined
        }
      };

      const decision = await this.stix.authorize(req);

      if (decision.decision === "DENY") return { kind: "DENIED", decision };
      if (decision.decision === "QUARANTINE") return { kind: "QUARANTINED", decision };

      const { args: newArgs, applied } = applyObligations(args, decision.obligations || []);

      try {
        const result = await handler(newArgs as any, decision.proof_token);

        // Outcome evidence (success)
        await this.stix.outcome({
          trace_id: decision.trace_id,
          status: "SUCCESS",
          result_summary: `tool=${meta.name}`,
          obligations_applied: applied,
          actor: this.base.actor,
          action: { name: meta.action, tool: meta.name, risk_tier: meta.riskTier },
          resource: { type: meta.resourceType, classification: req.resource.classification, attributes: resourceAttrs },
          context: this.base.context,
          policy: decision.policy
        });

        return { kind: "OK", value: result, trace_id: decision.trace_id };
      } catch (e: any) {
        await this.stix.outcome({
          trace_id: decision.trace_id,
          status: "FAILED",
          error_class: e?.name || "Error",
          result_summary: e?.message || "unknown",
          obligations_applied: applied,
          actor: this.base.actor,
          action: { name: meta.action, tool: meta.name, risk_tier: meta.riskTier },
          resource: { type: meta.resourceType, classification: req.resource.classification, attributes: resourceAttrs },
          context: this.base.context,
          policy: decision.policy
        });

        throw e;
      }
    };
  }
}

import { AuthorizeRequest, DecisionResponse } from "./types.js";

export class StixClient {
  constructor(private enforcerUrl: string) {}

  async authorize(req: AuthorizeRequest): Promise<DecisionResponse> {
    const r = await fetch(`${this.enforcerUrl}/authorize`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(req)
    });
    if (!r.ok) throw new Error(`Enforcer error ${r.status}: ${await r.text()}`);
    return (await r.json()) as DecisionResponse;
  }

  async outcome(payload: any): Promise<void> {
    const r = await fetch(`${this.enforcerUrl}/outcome`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(payload)
    });
    if (!r.ok) throw new Error(`Outcome error ${r.status}: ${await r.text()}`);
  }
}

type AuthorizeRequest = any;

function matchScope(req: AuthorizeRequest, p: any): boolean {
  return req.actor.type === p.scope.actor_type &&
         req.action === p.scope.action &&
         req.resource.type === p.scope.resource_type;
}

function matchConditions(req: AuthorizeRequest, p: any): boolean {
  const c = p.conditions || {};
  if (c.env && req.context.env !== c.env) return false;

  if (c.require_signal) {
    for (const [k, v] of Object.entries(c.require_signal)) {
      const actual = req.context?.signals?.[k];
      if (actual !== v) return false;
    }
  }

  if (c.require_resource_attr) {
    for (const [k, v] of Object.entries(c.require_resource_attr)) {
      const actual = req.resource?.attributes?.[k];
      if (actual !== v) return false;
    }
  }
  return true;
}

function decisionRank(d: string): number {
  if (d === "DENY") return 0;
  if (d === "QUARANTINE") return 1;
  return 2;
}

function evaluateSinglePolicy(req: AuthorizeRequest, p: any) {
  if (!matchScope(req, p)) return null;
  if (!matchConditions(req, p)) return null;
  return {
    decision: p.decision,
    reason_code: p.reason_code,
    policy_id: p.policy_id,
    version: p.version,
    priority: typeof p.priority === "number" ? p.priority : 100
  };
}

async function main() {
  const traceId = process.argv[2];
  if (!traceId) {
    console.error("Usage: stix-replay <trace_id>");
    process.exit(2);
  }

  const evidenceUrl = process.env.STIX_EVIDENCE_URL || "http://localhost:8083";
  const policyUrl = process.env.STIX_POLICY_URL || "http://localhost:8082";

  const ev = await fetch(`${evidenceUrl}/evidence/search?trace_id=${encodeURIComponent(traceId)}`);
  if (!ev.ok) throw new Error(`evidence search failed: ${ev.status} ${await ev.text()}`);
  const { events } = await ev.json();

  if (!events?.length) {
    console.error("No events found for trace:", traceId);
    process.exit(1);
  }

  // Use the first decision event that has policy.hash
  const decisionEvent = events.find((e: any) => e.policy?.hash && e.decision && e.reason_code);
  if (!decisionEvent) {
    console.error("No decision event with policy hash found.");
    process.exit(1);
  }

  const policyHash = decisionEvent.policy.hash;
  const pr = await fetch(`${policyUrl}/policy/by-hash/${encodeURIComponent(policyHash)}`);
  if (!pr.ok) throw new Error(`policy fetch failed: ${pr.status} ${await pr.text()}`);
  const { policy } = await pr.json();

  // Rebuild authorize request from evidence event snapshot
  const authorizeReq: AuthorizeRequest = {
    actor: decisionEvent.actor,
    action: decisionEvent.action?.name,
    resource: decisionEvent.resource,
    context: decisionEvent.context,
    tool_call: { tool: decisionEvent.action?.tool, risk_tier: decisionEvent.action?.risk_tier }
  };

  const res = evaluateSinglePolicy(authorizeReq, policy);
  if (!res) {
    console.error("Replay FAIL: policy no longer matches the captured inputs (unexpected).");
    process.exit(1);
  }

  const same = res.decision === decisionEvent.decision;

  console.log(JSON.stringify({
    trace_id: traceId,
    recorded: { decision: decisionEvent.decision, reason_code: decisionEvent.reason_code, policy_hash: policyHash },
    replayed: { decision: res.decision, reason_code: res.reason_code, policy_id: res.policy_id, version: res.version },
    match: same
  }, null, 2));

  process.exit(same ? 0 : 1);
}

main().catch(e => {
  console.error(e);
  process.exit(1);
});
